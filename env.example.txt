# RevifyRCM — server settings
#
# These belong in the hosting environment, never in a file that is served
# to a browser. On Netlify: Site configuration → Environment variables.
#
# Copy this file to .env for local function testing. Do not commit .env.
# ═══════════════════════════════════════════════════════════════════════

# ── The database ───────────────────────────────────────────────────────
# Required before switching _config.js to 'api'. A free Neon or Supabase
# project is enough. Run schema.sql then schema-data.sql against it once.
DATABASE_URL=postgresql://user:password@host/dbname?sslmode=require

# Signs session cookies. Any long random string; changing it signs
# everybody out. Generate one with:  openssl rand -base64 48
SESSION_SECRET=

# Used to sign the short-lived token between a password check and a
# two-step code. Falls back to SESSION_SECRET when unset.
JWT_SECRET=

# Allows the very first administrator to be created through /api/bootstrap.
# Only works while the accounts table is empty. Remove it afterwards.
BOOTSTRAP_SECRET=

# ── Stedi: claims, eligibility, remittances ────────────────────────────
# A PRODUCTION key. Test keys work for eligibility only and are refused by
# the claim endpoints.
# Stedi portal → your account → API Keys
STEDI_API_KEY=

# Must match the key: a production key requires P. Sending T with a
# production key is refused outright.
#   P  claims are transmitted to payers for payment
#   T  test keys only
STEDI_USAGE=P

# Optional. Some payers reject claims without a submitter identification.
STEDI_SUBMITTER_ID=

# Optional. Only set these to point at a different Stedi environment.
# STEDI_CLAIMS_URL=
# STEDI_STATUS_URL=
# STEDI_REPORTS_URL=

# ── Automatic nightly submission ───────────────────────────────────────
# The job wakes every hour and submits only when the local hour matches.
# Hourly rather than once a day so daylight saving never shifts the time.
CLAIMS_CRON_HOUR=21
CLAIMS_TZ=America/Chicago

# Remittances are collected an hour later, so the two jobs never overlap.
ERA_CRON_HOUR=22

# ── Email ──────────────────────────────────────────────────────────────
# For password reset links. Without it, reset emails are not sent.
RESEND_API_KEY=
MAIL_FROM=noreply@yourdomain.com

# ── Text messages for the patient portal ────────────────────────────────
# Patients sign in with a code sent to their phone. Without these the code is
# shown on screen instead, clearly marked as testing — fine while building,
# never for real patients.
# Any SMS provider works; this is written for Twilio.
TWILIO_ACCOUNT_SID=
TWILIO_AUTH_TOKEN=
TWILIO_FROM=
PRACTICE_NAME=

# ── Phone scheduling (Twilio Voice) ─────────────────────────────────────
# A patient can call in and book straight onto the Schedule calendar — see
# VOICE-SCHEDULING-SETUP.md. Reuses the TWILIO_* variables above: the same
# Account SID/Auth Token answer and verify the call, TWILIO_FROM sends the
# booking confirmation text.
# IANA time zone the phone system uses to decide what "today" and "the next
# opening" mean. Falls back to CLAIMS_TZ above, then America/Chicago.
SCHEDULING_TZ=America/Chicago

# ── Site ───────────────────────────────────────────────────────────────
# Your own address. Used in reset links and to restrict which origins the
# functions answer. Leaving it unset allows any origin, which is fine while
# testing and worth setting before real patient data.
SITE_URL=https://your-site.netlify.app

# Require two-step verification for every account.
REQUIRE_MFA=false

# ── Claim queue storage ────────────────────────────────────────────────
# Only needed if you use Netlify Blobs rather than the database. The queue
# prefers DATABASE_URL when it is set, so these can usually stay empty.
# Note: names beginning NETLIFY_ are reserved and never reach a function.
# RF_SITE_ID=
# RF_BLOBS_TOKEN=

# Or MongoDB instead of Postgres for the queue.
